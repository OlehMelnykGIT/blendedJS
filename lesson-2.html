Масив styles і функція logItems

pushindexOfarr[i] = …forfor...ofjs-inKlass/lesson-2.js

Умова

Створити масив styles з 'jazz' і 'blues', додати в кінець 'rock-n-roll', знайти 'blues' і замінити на 'classic'. Потім написати logItems(array), яка циклом for виводить "<номер> - <значення>", нумерація з 1. 

const styles = ["jazz", "blues"];
styles.push("rock-n-roll");

const index = styles.indexOf("blues");
if (index !== -1) {
  styles[index] = "classic";
}

function logItems(array) {
  for (let i = 0; i < array.length; i++) {
    console.log(`${i + 1} - ${array[i]}`);
  }
}

logItems(styles);

// for...of — коли індекс не потрібен
for (const style of styles) {
  console.log(style);
}

Як це працює

  • push додає елемент у кінець і змінює сам масив.
  • indexOf повертає індекс першого збігу або -1, якщо елемента немає. Тому перевіряємо index !== -1 перед заміною.
  • Індекси починаються з 0, тому для нумерації виводимо i + 1.
  • for дає доступ до індексу, а for...of коротший, коли потрібні лише значення.

checkLogin: чи є ім'я в масиві

includesfor...ofreturn у цикліjs-inKlass/lesson-2.js

Умова

checkLogin(array) питає ім'я через prompt. Якщо воно є в масиві, показує "Welcome, <name>!", інакше "User not found". 

const logins = ["Peter", "John", "Igor", "Sasha"];

function checkLogin(array) {
  const login = prompt("What is your name?");

  if (array.includes(login)) {
    alert(`Welcome, ${login}!`);
  } else {
    alert("User not found");
  }
}

// варіант з циклом: "не знайдено" — тільки ПІСЛЯ циклу
function checkLoginLoop(array) {
  const login = prompt("What is your name?");

  for (const name of array) {
    if (name === login) {
      alert(`Welcome, ${name}!`);
      return;
    }
  }
  alert("User not found");
}

Як це працює

  • includes сам перебирає масив і повертає true або false. Для перевірки «чи є» це найкоротший варіант.
  • У циклі return одразу виходить з функції, коли ім'я знайдено.
  • Висновок «не знайдено» можна зробити, лише коли переглянули весь масив, тобто після циклу.
Зауваження до коду в lesson-2.js
  • alert("User not found") стоїть всередині циклу. Для "Igor" ти побачиш «User not found» двічі (на Peter і John), а потім «Welcome». Подивись на це в демо.

Середнє значення довільної кількості аргументів

...argstypeof|| 0js-inKlass/lesson-2.js

Умова

calculateAverage() приймає будь-яку кількість аргументів і повертає їхнє середнє значення. Враховуються тільки числа. 

function calculateAverage(...args) {
  let sum = 0;
  let count = 0;

  for (const arg of args) {
    if (typeof arg === "number") {
      sum += arg;
      count += 1;
    }
  }

  return count === 0 ? 0 : sum / count;
}

console.log(calculateAverage(1, 2, 3));       // 2
console.log(calculateAverage("2", 2, 3));     // 2.5
console.log(calculateAverage("1", "2", "3")); // 0

Як це працює

  • Rest-параметр ...args збирає всі передані аргументи у справжній масив.
  • Рахуємо не args.length, а окремий count, бо рядки пропускаємо.
  • Без чисел вийшло б 0 / 0 = NaN. Твій варіант sum / count || 0 теж працює: NaN хибне, тому || поверне 0.
Зауваження до коду в lesson-2.js
  • Описка в назві: caclculateAverage → calculateAverage.

Сума сусідніх чисел

forarray[i + 1]length - 1js-inKlass/lesson-2.js

Умова

Скласти перше число з другим, друге з третім і так до кінця, результати зібрати в новий масив. Для [22, 11, 34, 5, 12, 13, 14, 15] вийде [33, 45, 39, 17, 25, 27, 29]. 

const someArr = [22, 11, 34, 5, 12, 13, 14, 15];

function sumNeighbours(array) {
  const result = [];

  for (let i = 0; i < array.length - 1; i++) {
    result.push(array[i] + array[i + 1]);
  }

  return result;
}

console.log(sumNeighbours(someArr)); // [33, 45, 39, 17, 25, 27, 29]

Як це працює

  • Пар на одну менше, ніж елементів, тому умова i < array.length - 1.
  • Якби дійшли до останнього i, то array[i + 1] було б undefined, і сума дала б NaN.
  • Оригінальний масив не змінюється, результат іде в новий result.

Найменше число в масиві

Array.isArrayfor...ofMath.min(...arr)js-inKlass/lesson-2.js

Умова

findSmallestNumber(numbers) повертає найменше число масиву. Якщо передали не масив, повертає 'Sorry, it is not an array!'. 

const numbers = [2, 5, 35, 56, 12, 24, 7, 80, 3];

function findSmallestNumber(numbers) {
  if (!Array.isArray(numbers)) {
    return "Sorry, it is not an array!";
  }

  let smallest = numbers[0];

  for (const number of numbers) {
    if (number < smallest) {
      smallest = number;
    }
  }

  return smallest; // або Math.min(...numbers)
}

console.log(findSmallestNumber(numbers)); // 2
console.log(findSmallestNumber("abc"));   // "Sorry, it is not an array!"

Як це працює

  • typeof [] повертає "object", тому масив перевіряємо через Array.isArray().
  • Беремо перший елемент як «поки що найменший» і порівнюємо з ним решту.
  • Коротший варіант: Math.min(...numbers). Spread ... розкладає масив на окремі аргументи.
Зауваження
  • У lesson-2.js тіло функції поки порожнє. Вище готове рішення.

Найдовше слово в реченні

split(" ")lengthfor...ofjs-inKlass/lesson-2.js

Умова

findLongestWord(string) приймає речення зі слів через пробіл і повертає найдовше слово. 

function findLongestWord(string) {
  const words = string.split(" ");
  let longest = "";

  for (const word of words) {
    if (word.length > longest.length) {
      longest = word;
    }
  }

  return longest;
}

console.log(findLongestWord("London is the capital of Great Britain")); // "capital"

Як це працює

  • split(" ") ділить рядок за пробілами і повертає масив слів.
  • Порівнюємо довжини, а зберігаємо саме слово.
  • Через строге > при однаковій довжині лишається перше знайдене слово.
Зауваження
  • У lesson-2.js цю задачу ще не розв'язано. Вище готове рішення.

Змінюємо об'єкт user і виводимо ключі

obj.key = …Object.keysObject.entriesobj[key]js-inKlass/lesson-2.js

Умова

Для об'єкта user: додати mood: 'happy', замінити hobby на 'skydiving' і premium на false. Потім вивести '<ключ>:<значення>' через Object.keys() і for...of. 

const user = {
  name: "John",
  age: 20,
  hobby: "tenis",
  premium: true,
};

user.mood = "happy";       // нового ключа не було, тому він додається
user.hobby = "skydiving";  // ключ уже був, тому значення замінюється
user.premium = false;

const keys = Object.keys(user);
for (const key of keys) {
  console.log(`${key}: ${user[key]}`);
}

// те саме через entries: кожен елемент — пара [ключ, значення]
for (const [key, value] of Object.entries(user)) {
  console.log(`${key}: ${value}`);
}

Як це працює

  • Присвоєння через крапку і додає новий ключ, і змінює наявний. const забороняє лише замінити весь об'єкт, але не його поля.
  • Object.keys повертає масив ключів, Object.values масив значень, Object.entries масив пар [ключ, значення].
  • Звернутись до властивості можна трьома способами: user.name, user["name"] і user[key], коли ключ лежить у змінній. У циклі підходить тільки останній: user.key шукав би ключ з буквальною назвою "key".

Сума зарплат команди

Object.values+=js-inKlass/lesson-2.js

Умова

Порахувати суму всіх зарплат з об'єкта salaries і зберегти в sum. Для порожнього об'єкта результат має бути 0. 

const salaries = {
  Mango: 100,
  Poly: 160,
  Ajax: 1470,
};

let sum = 0;

for (const salary of Object.values(salaries)) {
  sum += salary;
}

console.log(sum); // 1730

Як це працює

  • Імена тут не потрібні, тому беремо лише значення через Object.values.
  • Для порожнього об'єкта Object.values({}) повертає [], цикл не виконується жодного разу, і sum лишається 0.
Зауваження до коду в lesson-2.js
  • sum + -salary; лише обчислює вираз і нікуди його не записує, тому sum лишається 0. Потрібно sum += salary;.

sum + -salary

Об'єкт calculator і this

методиthisundefinedjs-inKlass/lesson-2.js

Умова

Об'єкт calculator з методами read(a, b) (зберігає аргументи у властивостях), sum(), mult() і exist() (перевіряє, що властивості є). Якщо їх немає, sum і mult повертають 'No such properties'. 

const calculator = {
  read(a, b) {
    this.a = a;
    this.b = b;
  },
  exist() {
    return this.a !== undefined && this.b !== undefined;
  },
  sum() {
    if (!this.exist()) {
      return "No such properties";
    }
    return this.a + this.b;
  },
  mult() {
    if (!this.exist()) {
      return "No such properties";
    }
    return this.a * this.b;
  },
};

console.log(calculator.sum()); // "No such properties"
calculator.read(5, 7);
console.log(calculator.sum());  // 12
console.log(calculator.mult()); // 35

Як це працює

  • this всередині методу — це об'єкт, на якому метод викликали, тобто calculator у calculator.read(…).
  • read створює властивості this.a і this.b. До першого виклику їх немає, і звернення до них дає undefined.
  • Порівняння !== вже повертає true або false, тому ? true : false зайве.
Зауваження до коду в lesson-2.js
  • У mult стоїть this.num1 = this.num2. Це присвоєння, а не множення: метод повертає num2 і ще й затирає num1. Потрібно this.num1 * this.num2.
  • Описка в рядку відповіді: propeties → properties.

calcTotalPrice: вартість фрукта

масив об'єктівfor...offruit.namejs-inKlass/lesson-2.js

Умова

calcTotalPrice(fruits, fruitName) повертає загальну вартість (price × quantity) усіх об'єктів з назвою fruitName. Об'єктів з однаковою назвою може бути кілька. 

const fruits = [
  { name: "Яблуко", price: 45, quantity: 7 },
  { name: "Апельсин", price: 60, quantity: 4 },
  { name: "Банан", price: 125, quantity: 8 },
  { name: "Груша", price: 350, quantity: 2 },
  { name: "Виноград", price: 440, quantity: 3 },
  { name: "Банан", price: 125, quantity: 3 },
];

function calcTotalPrice(fruits, fruitName) {
  let totalPrice = 0;

  for (const fruit of fruits) {
    if (fruit.name === fruitName) {
      totalPrice += fruit.price * fruit.quantity;
    }
  }

  return totalPrice;
}

console.log(calcTotalPrice(fruits, "Банан")); // 1375

Як це працює

  • Кожен елемент масиву — об'єкт, тому в циклі fruit має поля name, price і quantity.
  • Цикл не зупиняється на першому збігу. Він проходить увесь масив і додає вартість кожного збігу до totalPrice.
Пісочниці перераховують результат при кожній зміні поля. Код у блоках виправлений і відповідає умовам; зауваження стосуються версії в js-inKlass/lesson-2.js.
